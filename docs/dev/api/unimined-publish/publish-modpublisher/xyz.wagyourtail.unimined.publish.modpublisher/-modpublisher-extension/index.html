ModpublisherExtension

abstract class ModpublisherExtension @Inject constructor(objects: ObjectFactory)

uniminedPublishModpublisher { } — upload destination config, mirroring ModPublisher's own flat surface. The adapter feeds these into publisher { } when com.hypherionmc.modutils.modpublisher is applied.

API keys are Property<String> providers — resolve them from providers.gradleProperty/providers.environmentVariable; they are only read at execution time and never logged.

uniminedPublishModpublisher {
    apiKeys {
        curseforge.set(providers.environmentVariable("CURSE_TOKEN"))
        modrinth.set(providers.environmentVariable("MODRINTH_TOKEN"))
    }
    curseID.set("1234"); modrinthID.set("abc")
    versionType.set("release")
    mergedArtifact.set(layout.buildDirectory.file("libs/merged.jar"))
}
